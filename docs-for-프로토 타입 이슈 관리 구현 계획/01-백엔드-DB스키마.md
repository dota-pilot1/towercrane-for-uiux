## 추가할 테이블

### issuesTable

```ts
export const issuesTable = sqliteTable('issues', {
  id: text('id').primaryKey(),                          // 'issue-{uuid}'
  prototypeId: text('prototype_id')
    .notNull()
    .references(() => prototypesTable.id, { onDelete: 'cascade' }),
  title: text('title').notNull(),
  content: text('content').notNull().default(''),       // 최대 5000자
  issueType: text('issue_type')
    .$type<'BUG' | 'FEATURE' | 'IMPROVEMENT' | 'QUESTION' | 'OTHER'>()
    .notNull()
    .default('BUG'),
  status: text('status')
    .$type<'OPEN' | 'IN_PROGRESS' | 'TESTING' | 'CLOSED'>()
    .notNull()
    .default('OPEN'),
  priority: text('priority')
    .$type<'LOW' | 'MEDIUM' | 'HIGH' | 'URGENT'>()
    .notNull()
    .default('MEDIUM'),
  reporterId: text('reporter_id')
    .notNull()
    .references(() => usersTable.id, { onDelete: 'cascade' }),
  assigneeId: text('assignee_id')
    .references(() => usersTable.id, { onDelete: 'set null' }),
  dueDate: text('due_date'),                            // nullable
  orderIdx: integer('order_idx').notNull().default(0),
  createdAt: text('created_at').notNull(),
  updatedAt: text('updated_at').notNull(),
});
```

### issueCommentsTable

```ts
export const issueCommentsTable = sqliteTable('issue_comments', {
  id: text('id').primaryKey(),                          // 'issue-cmt-{uuid}'
  issueId: text('issue_id')
    .notNull()
    .references(() => issuesTable.id, { onDelete: 'cascade' }),
  userId: text('user_id')
    .notNull()
    .references(() => usersTable.id, { onDelete: 'cascade' }),
  content: text('content').notNull(),                   // 최대 2000자
  deleted: integer('deleted', { mode: 'boolean' }).notNull().default(false),
  createdAt: text('created_at').notNull(),
  updatedAt: text('updated_at').notNull(),
});
```

## Drizzle 마이그레이션

스키마 추가 후 다음 커맨드로 마이그레이션 파일 생성 및 적용:

```bash
# 마이그레이션 파일 생성
npx drizzle-kit generate

# 마이그레이션 적용 (서버 시작 시 자동 실행되는 방식 확인 필요)
npx drizzle-kit migrate
```

> 기존 Task 구조와 동일 패턴. `taskChecklistsTable`, `taskCommentsTable` 참고.
