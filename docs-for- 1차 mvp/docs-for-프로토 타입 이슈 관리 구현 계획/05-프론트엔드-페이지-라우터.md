## prototype-issues-page.tsx

```tsx
import { useSearch, useNavigate } from '@tanstack/react-router'
import { useState } from 'react'
import { ArrowLeft, Plus, RefreshCw } from 'lucide-react'
import { useIssues } from '../../../features/issue/model/use-issue-queries'
import { IssueKanbanView } from '../../../features/issue/ui/issue-kanban-view'
import { IssueTableView } from '../../../features/issue/ui/issue-table-view'
import { IssueFormDialog } from '../../../features/issue/ui/issue-form-dialog'
import { IssueDetailDialog } from '../../../features/issue/ui/issue-detail-dialog'
import { Button } from '../../../shared/ui/button'
import type { IssueFilters } from '../../../entities/issue/model/types'

type ViewMode = 'kanban' | 'table'

export function PrototypeIssuesPage() {
  const { prototypeId } = useSearch({ strict: false }) as { prototypeId?: string }
  const navigate = useNavigate()

  const [viewMode, setViewMode] = useState<ViewMode>('kanban')
  const [createOpen, setCreateOpen] = useState(false)
  const [selectedIssueId, setSelectedIssueId] = useState<string | null>(null)

  const [filters, setFilters] = useState<Omit<IssueFilters, 'prototypeId'>>({
    page: 1,
    pageSize: 50,
    sort: 'order',
  })

  const issuesQuery = useIssues({ prototypeId: prototypeId ?? '', ...filters })
  const issues = issuesQuery.data?.items ?? []

  if (!prototypeId) {
    return <div>프로토타입을 선택해주세요.</div>
  }

  return (
    <div className="flex flex-col gap-4">
      {/* 헤더 */}
      <div className="flex items-center justify-between">
        <div className="flex items-center gap-3">
          <button onClick={() => navigate({ to: -1 })} className="ui-icon-button">
            <ArrowLeft className="size-4" />
          </button>
          <div>
            <h1 className="text-xl font-black ui-text-primary">이슈 관리</h1>
            <p className="text-xs ui-text-muted">프로토타입 이슈를 추적하고 관리합니다</p>
          </div>
        </div>
        <div className="flex items-center gap-2">
          {/* 뷰 전환 토글 (칸반 / 테이블) */}
          <div className="flex rounded-sm border border-surface-border">
            <button
              onClick={() => setViewMode('kanban')}
              className={`px-3 py-1.5 text-xs font-bold transition-colors ${viewMode === 'kanban' ? 'bg-brand-glass text-brand-primary' : 'ui-text-muted'}`}
            >
              칸반
            </button>
            <button
              onClick={() => setViewMode('table')}
              className={`px-3 py-1.5 text-xs font-bold transition-colors ${viewMode === 'table' ? 'bg-brand-glass text-brand-primary' : 'ui-text-muted'}`}
            >
              테이블
            </button>
          </div>
          <button onClick={() => issuesQuery.refetch()} className="ui-icon-button">
            <RefreshCw className="size-4" />
          </button>
          <Button onClick={() => setCreateOpen(true)} size="sm">
            <Plus className="size-4 mr-1" />
            새 이슈
          </Button>
        </div>
      </div>

      {/* 상태별 카운트 뱃지 */}
      {/* OPEN N / IN_PROGRESS N / TESTING N / CLOSED N */}

      {/* 뷰 */}
      {viewMode === 'kanban' ? (
        <IssueKanbanView
          issues={issues}
          isLoading={issuesQuery.isLoading}
          onOpenDetail={setSelectedIssueId}
        />
      ) : (
        <IssueTableView
          issues={issues}
          isLoading={issuesQuery.isLoading}
          onOpenDetail={setSelectedIssueId}
        />
      )}

      {/* 다이얼로그 */}
      <IssueFormDialog
        prototypeId={prototypeId}
        open={createOpen}
        onOpenChange={setCreateOpen}
      />
      {selectedIssueId && (
        <IssueDetailDialog
          issueId={selectedIssueId}
          open={!!selectedIssueId}
          onOpenChange={(open) => !open && setSelectedIssueId(null)}
        />
      )}
    </div>
  )
}
```

## router.tsx 수정

```tsx
// 1. import 추가
import { PrototypeIssuesPage } from '../pages/prototype-issues/ui/prototype-issues-page'

// 2. 라우트 정의 추가
const prototypeIssuesRoute = createRoute({
  getParentRoute: () => appLayoutRoute,
  path: '/issues',
  validateSearch: (search: Record<string, unknown>) => ({
    prototypeId: typeof search.prototypeId === 'string' ? search.prototypeId : undefined,
  }),
  component: PrototypeIssuesPage,
})

// 3. routeTree에 추가
export const router = createRouter({
  routeTree: rootRoute.addChildren([
    loginRoute,
    appLayoutRoute.addChildren([
      indexRoute,
      prototypeIndexRoute,
      prototypeCategoryRoute,
      chatbotRoute,
      meetingRoute,
      docuRoute,

      apiDocRoute,
      taskRoute,

      adminUsersRoute,

  ]),
})
```
