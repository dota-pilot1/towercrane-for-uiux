// chatbot.service.ts
createSession(title?: string) {
  const now = new Date().toISOString()
  const session = {
    id: randomUUID(),
    title: title?.trim() || '새 대화',
    createdAt: now,
    updatedAt: now,
  }
  this.db.insert(chatSessionsTable).values(session).run()
  return session       // 생성된 객체 그대로 반환
}
포인트:
  • 프론트가 보내는 body는 빈 객체 — title은 서버가 '새 대화'로 기본값 설정
  • id는 서버가 UUID로 생성 → 응답에 담아 반환
  • 프론트는 응답 받자마자 store에 추가 + 활성화

renameSession(id: string, title: string) {
  const trimmed = title.trim()
  if (!trimmed) throw new NotFoundException('title is required')
  const now = new Date().toISOString()
  const result = this.db
    .update(chatSessionsTable)
    .set({ title: trimmed, updatedAt: now })
    .where(eq(chatSessionsTable.id, id))
    .run()
  if (result.changes === 0) {
    throw new NotFoundException('session not found')
  }
  return { id, title: trimmed, updatedAt: now }
}
포인트:
  • title뿐 아니라 updatedAt도 갱신 → 목록 정렬 순서가 바뀜
  • result.changes === 0 체크로 존재하지 않는 id 거름

③ ④ 메시지 저장 — 핵심 로직

메시지 저장은 별도 API가 없다. POST /chatbot/stream 안에서 자동으로 저장된다.
user 메시지는 스트리밍 시작 직전, assistant 메시지는 스트리밍 완료 직후.

[백엔드] streamGpt — DB 저장 + SSE 응답
async streamGpt(sessionId: string, message: string, res: Response) {
  // (1) 세션 갱신 + 첫 메시지면 title 자동 설정
  const meta = this.touchSession(sessionId, message)

  // (2) ★ user 메시지 DB 저장
  const userMessage = this.insertMessage(sessionId, 'user', message)

  // (3) meta 이벤트 전송 (진짜 id 알림)
  res.write(`data: ${JSON.stringify({
    type: 'meta', userMessage: userMessage, sessionTitle: meta.title,
  })}\n\n`)

  // (4) OpenAI 스트리밍 + 누적
  let assistantContent = ''
  for await (const chunk of stream) {
    const text = chunk.choices[0]?.delta?.content ?? ''
    if (text) {
      assistantContent += text                      // 변수에 누적만
      res.write(`data: ${JSON.stringify({ text })}\n\n`)
    }
  }

  // (5) ★ assistant 메시지 DB 저장 (완성본 한 번에)
  const assistantMessage = this.insertMessage(sessionId, 'assistant', assistantContent)

  // (6) done 이벤트 + [DONE]
  res.write(`data: ${JSON.stringify({ type: 'done', assistantMessage: assistantMessage })}\n\n`)
  res.write('data: [DONE]\n\n')
  res.end()
}

insertMessage 헬퍼

private insertMessage(sessionId, role, content) {
  const message = {
    id: randomUUID(),
    sessionId,
    role,        // 'user' or 'assistant'
    content,
    createdAt: new Date().toISOString(),
  }
  this.db.insert(chatMessagesTable).values(message).run()
  return message
}
왜 assistant는 청크마다 저장하지 않나?

청크마다 INSERT/UPDATE 하면:
  • 토큰 하나에 DB 쓰기 한 번 → 수십~수백 번 발생
  • SQLite도 부담, 디스크 I/O 증가
대신 메모리 변수 assistantContent에 누적했다가 완성본을 한 번만 INSERT.
